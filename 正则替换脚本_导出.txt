<body>
<script>
(function(){
  if(document.getElementById('app')&&document.getElementById('app').classList.contains('on')){return;}
  fetch('https://cdn.jsdelivr.net/gh/cj15000950009-dotcom/---@46cfd5a/index.html').then(r=>r.text()).then(h=>{
    var doc=new DOMParser().parseFromString(h,'text/html');
    var sts=doc.querySelectorAll('style');
    for(var i=0;i<sts.length;i++){var st=document.createElement('style');st.textContent=sts[i].textContent;document.head.appendChild(st);}
    document.body.innerHTML=doc.body.innerHTML;
    var scs=doc.body.querySelectorAll('script');
    for(var j=0;j<scs.length;j++){var ns=document.createElement('script');if(scs[j].src){ns.src=scs[j].src;}else{ns.textContent=scs[j].textContent;}document.body.appendChild(ns);}
  })
})();
</script>
</body>
